// Companion Mobile App: Prince Patel Smart Link & Flower QR (Expo / React Native)
import React, { useState, useEffect } from 'react';
import { View, Text, TextInput, TouchableOpacity, FlatList, Share, Linking } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { CameraView, useCameraPermissions } from 'expo-camera';
import * as Notifications from 'expo-notifications';

const API_BASE = 'https://pp.link/api/v1';
export default function PrinceSmartQRApp() {
  const [longUrl, setLongUrl] = useState('');
  const [alias, setAlias] = useState('');
  const [history, setHistory] = useState([]);
  const [scanning, setScanning] = useState(false);

  useEffect(() => {
    AsyncStorage.getItem('PP_MOBILE_LINKS').then(d => d && setHistory(JSON.parse(d)));
  }, []);

  const createShortLink = async () => {
    const slug = alias || Math.random().toString(36).substring(2, 7);
    const entry = { id: Date.now().toString(), longUrl, shortUrl: `https://pp.link/${slug}`, clicks: 0 };
    const updated = [entry, ...history];
    setHistory(updated);
    await AsyncStorage.setItem('PP_MOBILE_LINKS', JSON.stringify(updated));
  };

  return (
    <View style={{ flex: 1, backgroundColor: '#060913', padding: 20 }}>
      <Text style={{ color: '#14b8a6', fontSize: 22, fontWeight: 'bold' }}>👑 PP.LINK Mobile Studio</Text>
    </View>
  );
}
